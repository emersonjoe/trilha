# Contrato: CLI `trilha`

```
trilha new <dir> [--module github.com/x/y]   cria projeto; módulo padrão = nome da pasta
trilha gen                                    gera trilha_gen.go a partir de app/
trilha dev [--addr :3000]                     gen + build + run + watch + live-reload
trilha build [-o bin/<nome>]                  gen + go build (CGO_ENABLED=0, -trimpath)
trilha routes                                 tabela: MÉTODOS  PADRÃO  ORIGEM
trilha version
```

Saídas de erro do gerador vão para stderr com código 1, uma linha por violação.
`dev` imprime `→ http://localhost:3000` ao subir, `↻ rebuild (123 ms)` a cada ciclo e o
erro do compilador quando falhar. Variáveis: `TRILHA_ENV` (dev|prod), `PORT`/`ADDR`.
`trilha_gen.go` gerado:

```go
// Code generated by trilha. DO NOT EDIT.

package main

import (
	"embed"

	app "example.com/blog/app"
	app_admin "example.com/blog/app/admin"
	app_api_posts "example.com/blog/app/api/posts"
	app_api_posts_id_ "example.com/blog/app/api/posts/id_"
	app_blog "example.com/blog/app/blog"
	app_blog_novo "example.com/blog/app/blog/novo"
	app_blog_slug_ "example.com/blog/app/blog/slug_"
	app_docs_path__ "example.com/blog/app/docs/path__"
	trilha "github.com/emersonjoe/trilha"
)

//go:embed public
var publicFS embed.FS

// newApp builds the application. Tests can call it and use a.Handler().
func newApp() *trilha.App {
	cfg := trilha.ConfigFromEnv()
	cfg.Public = trilha.PublicFS(publicFS, "public")
	a := trilha.New(cfg)
	if err := app.Setup(a); err != nil {
		trilha.Fatal(err)
	}
	a.SetRootLayout(app.Layout)
	a.SetNotFound(app.NotFound)
	a.SetErrorPage(app.Error)
	a.Register(trilha.Route{
		Pattern:     "/",
		Page:        app.Page,
		Layouts:     []trilha.LayoutFunc{app.Layout},
		Middlewares: []trilha.MiddlewareFunc{app.Middleware},
	})
	a.Register(trilha.Route{
		Pattern:     "/admin",
		Page:        app_admin.Page,
		Layouts:     []trilha.LayoutFunc{app.Layout},
		Middlewares: []trilha.MiddlewareFunc{app.Middleware, app_admin.Middleware},
	})
	a.Register(trilha.Route{
		Pattern: "/api/posts",
		Methods: map[string]trilha.HandlerFunc{
			"DELETE": app_api_posts.DELETE,
			"GET":    app_api_posts.GET,
			"POST":   app_api_posts.POST,
		},
		Middlewares: []trilha.MiddlewareFunc{app.Middleware},
	})
	a.Register(trilha.Route{
		Pattern: "/api/posts/{id}",
		Methods: map[string]trilha.HandlerFunc{
			"GET": app_api_posts_id_.GET,
		},
		Middlewares: []trilha.MiddlewareFunc{app.Middleware},
	})
	a.Register(trilha.Route{
		Pattern:     "/blog",
		Page:        app_blog.Page,
		Layouts:     []trilha.LayoutFunc{app_blog.Layout, app.Layout},
		Middlewares: []trilha.MiddlewareFunc{app.Middleware},
	})
	a.Register(trilha.Route{
		Pattern: "/blog/novo",
		Page:    app_blog_novo.Page,
		Methods: map[string]trilha.HandlerFunc{
			"POST": app_blog_novo.POST,
		},
		Layouts:     []trilha.LayoutFunc{app_blog.Layout, app.Layout},
		Middlewares: []trilha.MiddlewareFunc{app.Middleware},
	})
	a.Register(trilha.Route{
		Pattern:     "/blog/{slug}",
		Page:        app_blog_slug_.Page,
		Layouts:     []trilha.LayoutFunc{app_blog.Layout, app.Layout},
		Middlewares: []trilha.MiddlewareFunc{app.Middleware},
	})
	a.Register(trilha.Route{
		Pattern:     "/docs/{path...}",
		Page:        app_docs_path__.Page,
		Layouts:     []trilha.LayoutFunc{app.Layout},
		Middlewares: []trilha.MiddlewareFunc{app.Middleware},
	})
	return a
}

func main() {
	trilha.Fatal(newApp().ListenAndServe())
}
```
